       ```rust
       fn encode_webp_lossless(img: &DynamicImage) -> Result<Vec<u8>> {
           let mut buf = Vec::new();
           let encoder = WebPEncoder::new_lossless(&mut buf);
           img.write_with_encoder(encoder)

           Ok(buf)
       }
       ```
