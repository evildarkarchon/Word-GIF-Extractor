```rust
/// Target format for image conversion
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum OutputFormat {
    Jpg,
    Png,
    Webp,
}

/// Returns true for decodable extensions: jpg, jpeg, png, gif, bmp, tiff, tif, webp, ico
/// Returns false for svg, wmf, emf
pub fn can_convert(extension: &str) -> bool;

/// Returns Ok(Some(bytes)) on success, Ok(None) for unsupported, Err for corrupt
pub fn convert_image(data: &[u8], format: OutputFormat, quality: u8) -> Result<Option<Vec<u8>>>;
```

       ```rust
       pub fn try_convert(
           data: &[u8],
           source_ext: &str,
           format: OutputFormat,
           quality: u8,
       ) -> Result<ConversionResult> {
           if !can_convert(source_ext) {
               return Ok(ConversionResult::Skipped(source_ext.to_lowercase()));
           }
           match convert_image(data, format, quality)? {
               Some(converted_bytes) => {
                   Ok(ConversionResult::Converted(
                       converted_bytes,
                       format.extension().to_string(),
                   ))
               }
               None => {
                   Ok(ConversionResult::Skipped(source_ext.to_lowercase()))
               }
           }
       }
       ```
